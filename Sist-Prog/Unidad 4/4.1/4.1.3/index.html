Objetivo

Tres ideas que sostienen todo

Juntas las clases de 4.1.1 y 4.1.2 y les agregas reglas: así se programa una máquina que no se lastima a sí misma ni a quien la usa.

🧩

Driver

Una clase que esconde pines y tiempos, y ofrece verbos: arrancar(), parar().

🔗

Interfaz común

Todos los actuadores heredan de Actuador: cualquiera se puede detener con la misma llamada.

🔒

Interbloqueo

Una regla que impide combinaciones peligrosas, como arrancar la banda con la guarda abierta.

Diseño

Clases y reglas

Interbloqueo no mueve pines: solo decide si una orden se permite y se la pasa al driver.

OrdenSe permite si…Si no…
ARRANCARla guarda está cerrada y quieta, y el motor ya se detuvo por completose rechaza + 3 parpadeos rojos
ABRIR GUARDAel motor está detenido (no en marcha ni frenando)se rechaza + 3 parpadeos rojos
CERRAR GUARDAsiempre—
PARARsiempre: el paro nunca se bloquea—
EMERGENCIAsiempre (PARAR sostenido 1 s): frena el motor y congela la guarda—
PRINCIPIO 1

Estado seguro por defecto

Cada begin() deja su actuador quieto y sin energía.

PRINCIPIO 2

Parar nunca falla

Arrancar puede ser rechazado; parar, jamás.

PRINCIPIO 3

Rechazo explícito

Si algo no se permite, se avisa con el motivo. Nada falla en silencio.

Simulador

Intenta romper las reglas

Misma lógica que el sketch: prueba arrancar con la guarda abierta o abrirla con la banda en marcha.

Celda: banda + guarda

    Boton (de la práctica 4.1.2, sin cambios)
      Boton.h
    // ===== Driver de un pulsador con filtro de rebote (sin delay) =====
class Boton {
  public:
    Boton(uint8_t pin, unsigned long estableMs = 25) : _pin(pin), _estableMs(estableMs) {}

    void begin() {
      pinMode(_pin, INPUT_PULLUP);              // suelto = HIGH, presionado = LOW
      _cruda = _estado = (digitalRead(_pin) == HIGH);
      _tCambio = millis();
    }

    // Llamar en CADA vuelta de loop(). Nunca usa delay().
    void actualizar() {
      _bajada = _subida = false;
      bool lectura = (digitalRead(_pin) == HIGH);
      if (lectura != _cruda) {                  // la señal cambió: puede ser rebote
        _cruda = lectura;
        _tCambio = millis();                    // reinicia el cronómetro de estabilidad
        if (!lectura) _caidasCrudas++;
      }
      if (lectura != _estado && millis() - _tCambio >= _estableMs) {
        _estado = lectura;                      // estable el tiempo suficiente: se acepta
        if (!_estado) { _bajada = true; _tPresion = millis(); }
        else          { _subida = true; }
      }
    }

    bool sePresiono()     const { return _bajada; }
    bool seSolto()        const { return _subida; }
    bool estaPresionado() const { return !_estado; }
    unsigned long tiempoPresionado() const { return estaPresionado() ? millis() - _tPresion : 0; }
    unsigned long caidasCrudas()     const { return _caidasCrudas; }

  private:
    uint8_t _pin;
    unsigned long _estableMs, _tCambio = 0, _tPresion = 0, _caidasCrudas = 0;
    bool _cruda = true, _estado = true, _bajada = false, _subida = false;   // true = HIGH
};
    Actuador · la interfaz
      Actuador.h
    // ===== Interfaz común: todo actuador sabe pararse =====
class Actuador {
  public:
    virtual void begin() = 0;
    virtual void parar() = 0;            // obligatorio y NUNCA se bloquea
    virtual void actualizar() = 0;       // avanza sin delay(); llamar en cada loop()
    virtual bool enReposo() const = 0;   // true = no se está moviendo
};

    = 0 significa "obligatorio": una clase que herede y no implemente parar() no compila.

    MotorDC · freno y luego libre
      MotorDC.h
    // ===== Driver del motor DC (L298N / L293D) =====
class MotorDC : public Actuador {
  public:
    enum class Estado { PARADO, MARCHA, FRENANDO };

    MotorDC(uint8_t en, uint8_t in1, uint8_t in2, uint8_t velocidad)
      : _en(en), _in1(in1), _in2(in2), _vel(velocidad) {}

    void begin() override {
      pinMode(_en, OUTPUT);
      pinMode(_in1, OUTPUT);
      pinMode(_in2, OUTPUT);
      _salidas(false, false, 0);             // estado seguro: sin energía
    }

    bool arrancar() {
      if (_estado != Estado::PARADO) return false;   // no re-arrancar mientras frena
      _salidas(true, false, _vel);
      _estado = Estado::MARCHA;
      return true;
    }

    void parar() override {
      if (_estado != Estado::MARCHA) return;  // ya parado o frenando
      _salidas(true, true, 255);              // freno dinámico
      _estado = Estado::FRENANDO;
      _t = millis();
    }

    void actualizar() override {
      if (_estado == Estado::FRENANDO && millis() - _t >= FRENO_MS) {
        _salidas(false, false, 0);            // libera el puente H
        _estado = Estado::PARADO;
      }
    }

    bool enReposo() const override { return _estado == Estado::PARADO; }
    Estado estado() const { return _estado; }

  private:
    static constexpr unsigned long FRENO_MS = 400;
    uint8_t _en, _in1, _in2, _vel;
    Estado _estado = Estado::PARADO;
    unsigned long _t = 0;

    void _salidas(bool a, bool b, uint8_t pwm) {
      analogWrite(_en, 0);
      digitalWrite(_in1, a ? HIGH : LOW);
      digitalWrite(_in2, b ? HIGH : LOW);
      analogWrite(_en, pwm);
    }
};
    Compuerta · servo que se mueve sin bloquear
      Compuerta.h
    // ===== Driver de la guarda (servomotor) =====
#include <Servo.h>

class Compuerta : public Actuador {
  public:
    Compuerta(uint8_t pin, uint8_t angCerrada, uint8_t angAbierta)
      : _pin(pin), _cerrada(angCerrada), _abierta(angAbierta) {}

    void begin() override {
      _actual = _objetivo = _cerrada;
      _servo.attach(_pin);
      _servo.write(_actual);                  // de inmediato a la posición segura
    }

    void abrir()  { _objetivo = _abierta; }
    void cerrar() { _objetivo = _cerrada; }
    void parar() override { _objetivo = _actual; }   // se congela donde esté

    void actualizar() override {              // 1° cada 15 ms: movimiento suave
      if (_actual == _objetivo || millis() - _t < PASO_MS) return;
      _t = millis();
      if (_objetivo > _actual) _actual++; else _actual--;
      _servo.write(_actual);
    }

    bool enReposo()    const override { return _actual == _objetivo; }
    bool estaCerrada() const { return enReposo() && _actual == _cerrada; }
    bool vaACerrar()   const { return _objetivo == _cerrada; }

  private:
    static constexpr unsigned long PASO_MS = 15;
    Servo _servo;
    uint8_t _pin, _cerrada, _abierta, _actual = 0, _objetivo = 0;
    unsigned long _t = 0;
};

    El servo no avisa su posición. Moviéndolo de 1° en 1° el programa siempre sabe dónde está, y puede congelarlo a medio camino.

    Interbloqueo · las reglas
      Interbloqueo.h
    // ===== Interbloqueo: decide qué órdenes son seguras =====
class Interbloqueo {
  public:
    Interbloqueo(MotorDC& motor, Compuerta& guarda)
      : _motor(motor), _guarda(guarda), _todos{&motor, &guarda} {}

    bool pedirArranque() {
      if (!_guarda.estaCerrada()) return _rechazo(F("ARRANCAR: la guarda no esta cerrada"));
      if (!_motor.arrancar())     return _rechazo(F("ARRANCAR: el motor aun no se detiene"));
      Serial.println(F("OK  motor en marcha"));
      return true;
    }

    bool pedirAbrirGuarda() {
      if (!_motor.enReposo()) return _rechazo(F("ABRIR: el motor sigue en movimiento"));
      _guarda.abrir();
      Serial.println(F("OK  abriendo guarda"));
      return true;
    }

    void pedirCerrarGuarda() {               // cerrar siempre es seguro
      _guarda.cerrar();
      Serial.println(F("OK  cerrando guarda"));
    }

    void paro() {                            // el paro nunca se rechaza
      _motor.parar();
      Serial.println(F("PARO"));
    }

    void paroEmergencia() {
      for (Actuador* a : _todos) a->parar(); // polimorfismo: cada uno sabe pararse
      Serial.println(F("!!! PARO DE EMERGENCIA"));
    }

  private:
    MotorDC& _motor;
    Compuerta& _guarda;
    Actuador* _todos[2];

    bool _rechazo(const __FlashStringHelper* motivo) {
      Serial.print(F("RECHAZADO  "));
      Serial.println(motivo);
      return false;
    }
};

    Celda con guarda de seguridad

    Una banda (motor DC) que solo puede funcionar con la guarda (servo) cerrada.

    AlimentaciónGNDSeñalMotor
    ComponenteConexión
    L298N ENA / IN1 / IN2D5 (sin jumper) / D8 / D7
    L298N OUT1 / OUT2Motor DC
    L298N +12V · GNDFuente externa 7–12 V · GND común con el Arduino
    Servo señal (naranja)D9
    Servo + (rojo) · − (café)5 V de fuente externa · GND común (en Tinkercad: 5V del Arduino)

    ¿Por qué fuente aparte para el servo? Su arranque, sumado al del motor, es el escenario clásico de brownout del tema 4.1.2.
      Programa principal
    // ===== Práctica 4.1.3 — Programa principal =====
// Pega ANTES de esto: Boton, Actuador, MotorDC, Compuerta e Interbloqueo.
const uint8_t PIN_ENA = 5, PIN_IN1 = 8, PIN_IN2 = 7, PIN_SERVO = 9;
const uint8_t BTN_ARRANCAR = 2, BTN_PARAR = 3, BTN_GUARDA = 4;
const uint8_t LED_VERDE = 12, LED_ROJO = 11;

MotorDC      banda(PIN_ENA, PIN_IN1, PIN_IN2, 180);
Compuerta    guarda(PIN_SERVO, 0, 90);            // 0° = cerrada, 90° = abierta
Interbloqueo seguridad(banda, guarda);
Boton bArrancar(BTN_ARRANCAR), bParar(BTN_PARAR), bGuarda(BTN_GUARDA);

uint8_t parpadeos = 0;                            // parpadeos pendientes del LED rojo
unsigned long tLed = 0;
bool ledRojo = false, emergenciaHecha = false;

void avisarRechazo(uint8_t veces) { parpadeos = veces; }

void setup() {
  pinMode(LED_VERDE, OUTPUT);
  pinMode(LED_ROJO, OUTPUT);
  banda.begin();                                  // 1) motor sin energía
  guarda.begin();                                 // 2) guarda en posición cerrada conocida
  bArrancar.begin(); bParar.begin(); bGuarda.begin();
  Serial.begin(115200);
  Serial.println(F("ARRANCAR=D2  PARAR=D3 (1 s = emergencia)  GUARDA=D4"));
}

void loop() {
  // 1) Leer entradas
  bArrancar.actualizar(); bParar.actualizar(); bGuarda.actualizar();

  // 2) Toda orden pasa por el interbloqueo
  if (bArrancar.sePresiono() && !seguridad.pedirArranque()) avisarRechazo(3);

  if (bParar.sePresiono()) seguridad.paro();
  if (bParar.tiempoPresionado() >= 1000 && !emergenciaHecha) {
    seguridad.paroEmergencia();
    emergenciaHecha = true;
    avisarRechazo(6);
  }
  if (bParar.seSolto()) emergenciaHecha = false;

  if (bGuarda.sePresiono()) {
    if (guarda.vaACerrar()) { if (!seguridad.pedirAbrirGuarda()) avisarRechazo(3); }
    else seguridad.pedirCerrarGuarda();
  }

  // 3) Cada driver avanza por su cuenta (sin delay)
  banda.actualizar();
  guarda.actualizar();

  // 4) Indicadores
  digitalWrite(LED_VERDE, banda.estado() == MotorDC::Estado::MARCHA ? HIGH : LOW);
  if (parpadeos > 0 && millis() - tLed >= 150) {
    tLed = millis();
    ledRojo = !ledRojo;
    digitalWrite(LED_ROJO, ledRojo ? HIGH : LOW);
    if (!ledRojo) parpadeos--;
  }
}
